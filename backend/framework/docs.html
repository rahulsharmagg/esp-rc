mini-express

A small Express-style framework. Plain JavaScript, CommonJS, class-based, no dependencies.

Getting started

Copy the folder into your project and require it.

const mini = require('./mini-express');

const app = mini();
app.get('/', (req, res) => res.send('Hello'));
app.listen(3000);

Routing

app.get('/users', handler);
app.post('/users', handler);
app.put('/users/:id', handler);
app.patch('/users/:id', handler);
app.delete('/users/:id', handler);
app.all('/any', handler);          // every method

Path patterns

PatternMatchesResult
/users/:id/users/7req.params.id = '7'
/posts/:slug?/posts and /posts/hellooptional param
/files/*/files/a/b.txtreq.params['*'] = 'a/b.txt'

Chain methods on one path

app.route('/items')
  .get((req, res) => res.json([]))
  .post((req, res) => res.status(201).json(req.body));

Multiple handlers

app.get('/secret', checkLogin, checkAdmin, (req, res) => res.send('ok'));

Routers & groups

const { Router } = require('./mini-express');

const users = new Router();
users.get('/', (req, res) => res.json([]));
users.get('/:id', (req, res) => res.json({ id: req.params.id }));

app.use('/users', users);

// or inline
app.group('/api/v1', (api) => {
  api.get('/ping', (req, res) => res.json({ pong: true }));
});

Middleware

A middleware receives (req, res, next). Call next() to continue, or send a response to stop.

1. Function

app.use((req, res, next) => {
  console.log(req.method, req.url);
  next();
});

2. Class

const { Middleware } = require('./mini-express');

class Auth extends Middleware {
  constructor(token) { super(); this.token = token; }
  handle(req, res, next) {
    if (req.get('authorization') !== 'Bearer ' + this.token) return res.sendStatus(401);
    next();
  }
}

app.use('/admin', Auth.create('secret'));   // returns a function
app.use('/admin', new Auth('secret'));      // objects with handle() also work

3. Plugin

A function that receives the whole app. Use it to add routes, middleware or properties.

function health(app, opts = {}) {
  app.get(opts.path || '/health', (req, res) => res.json({ ok: true }));
}
app.plugin(health, { path: '/status' });

Body parsing

app.use(mini.json({ limit: '2mb' }));   // default limit 1mb
app.use(mini.urlencoded());

app.post('/items', (req, res) => res.json(req.body));

Invalid JSON gives a 400 error. A body over the limit gives 413. The raw buffer is kept in req.rawBody.

Custom format:

const { createParser } = mini.bodyParser;
const xml = createParser('xml', (buf) => myXmlParse(buf.toString()), {});
app.use(xml());

Static files

const path = require('path');

app.use(mini.static(path.join(__dirname, 'public')));
app.use('/assets', mini.static(path.join(__dirname, 'public'), { index: 'home.html' }));

Add file types in lib/utils/mime.js.

WebSocket

app.ws('/chat/:room', (socket, req) => {
  socket.room = req.params.room;
  socket.json({ welcome: socket.room });

  socket.on('message', (msg) => {
    // send to everyone else in the room
    app.wss.broadcast(msg, (c) => c !== socket && c.room === socket.room);
  });

  socket.on('close', () => console.log('left'));
});
APIDescription
socket.send(data)Send string or Buffer
socket.json(obj)Send an object as JSON
socket.close(code)Close the connection
socket.on('message' | 'close' | 'error', fn)Events
app.wss.clientsSet of all connected sockets
app.wss.broadcast(data, filter?)Send to all, or to those where filter(client) is true

Browser test:

const ws = new WebSocket('ws://localhost:3000/chat/general');
ws.onmessage = (e) => console.log(e.data);
ws.onopen = () => ws.send('hello');

Error handling

Errors thrown in handlers, including async ones, go to the error handler automatically.

app.get('/boom', async () => {
  throw Object.assign(new Error('Bad input'), { status: 400 });
});

// error middleware: 4 arguments, register after your routes
app.use((err, req, res, next) => {
  res.status(err.status || 500).json({ error: err.message });
});

Override the defaults:

app.onNotFound = (req, res) => res.status(404).json({ error: 'Not found' });
app.onError = (err, req, res) => res.status(500).send('Oops');
